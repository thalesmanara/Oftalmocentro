const buscar = node({
  type: 'n8n-nodes-base.postgres',
  version: 2.6,
  config: {
    name: 'Buscar versão',
    credentials: pgCred,
    parameters: {
      operation: 'executeQuery',
      query:
        "SELECT\n" +
        "  dv.id,\n" +
        "  dv.document_id AS \"documentId\",\n" +
        "  dv.version_number AS \"versionNumber\",\n" +
        "  dv.is_current AS \"isCurrent\",\n" +
        "  dv.status,\n" +
        "  dv.file_name AS \"fileName\",\n" +
        "  dv.file_size AS \"fileSize\",\n" +
        "  dv.mime_type AS \"mimeType\",\n" +
        "  dv.title_snapshot AS \"titleSnapshot\",\n" +
        "  dv.description_snapshot AS \"descriptionSnapshot\",\n" +
        "  dv.sector_id_snapshot AS \"sectorIdSnapshot\",\n" +
        "  dv.category_id_snapshot AS \"categoryIdSnapshot\",\n" +
        "  dv.subcategory_id_snapshot AS \"subcategoryIdSnapshot\",\n" +
        "  dv.responsible_user_id_snapshot AS \"responsibleUserIdSnapshot\",\n" +
        "  dv.expiration_date AS \"expirationDate\",\n" +
        "  dv.processing_status AS \"processingStatus\",\n" +
        "  dv.created_by AS \"createdBy\",\n" +
        "  u.name AS \"createdByName\",\n" +
        "  dv.created_at AS \"createdAt\",\n" +
        "  LEFT(dv.checksum, 12) AS \"checksum\",\n" +
        "  (SELECT COUNT(*)::int FROM document_chunks dc WHERE dc.document_version_id = dv.id) AS \"chunkCount\"\n" +
        "FROM document_versions dv\n" +
        "JOIN documents d ON d.id = dv.document_id AND d.deleted_at IS NULL\n" +
        "LEFT JOIN users u ON u.id = dv.created_by\n" +
        "WHERE dv.document_id = '{{ $json.query.documentId }}'::uuid\n" +
        "  AND dv.id = '{{ $json.query.versionId }}'::uuid\n" +
        "LIMIT 1;",
      options: {},
    },
    output: [{ json: { id: 'v1', documentId: 'd1', versionNumber: 1, isCurrent: true, status: 'CURRENT', createdAt: '2026-01-01', chunkCount: 0 } }],
  },
});

const montar = node({
  type: 'n8n-nodes-base.code',
  version: 2,
  config: {
    name: 'Montar data',
    parameters: {
      mode: 'runOnceForAllItems',
      language: 'javaScript',
      jsCode:
        "const rows = $input.all().map((i) => i.json).filter((j) => j && j.id);\n" +
        "const norm = $('Normalizar request').first().json;\n" +
        "if (!rows.length) {\n" +
        "  return [{ json: { code: 'NOT_FOUND', message: 'Versão do documento não encontrada.', statusCode: 404, requestId: norm.requestId, requestStartedAtMs: norm.requestStartedAtMs, method: norm.method, path: norm.path, notFound: true } }];\n" +
        "}\n" +
        "let userId = ''; let sessionId = '';\n" +
        "try { const auth = $('Validar auth').first().json; userId = auth.userId || ''; sessionId = auth.sessionId || ''; } catch (_) {}\n" +
        "return [{ json: { data: rows[0], asList: false, statusCode: 200, requestId: norm.requestId, requestStartedAtMs: norm.requestStartedAtMs, method: norm.method, path: norm.path, userId, sessionId, notFound: false } }];",
    },
    output: [{ json: { data: { id: 'v1' }, asList: false, statusCode: 200, requestId: 'req-1', notFound: false } }],
  },
});

export default workflow('get-document-version-detail', 'GET Document Version Detail')
  .add(webhook)
  .to(normalizar)
  .to(validarAuth)
  .to(
    authOk
      .onTrue(
        validarPerm.to(
          permOk
            .onTrue(
              restaurar.to(
                buscar.to(
                  montar.to(
                    foundOk
                      .onTrue(sucesso.to(respond))
                      .onFalse(err404.to(respond404))
                  )
                )
              )
            )
            .onFalse(err403.to(respond403))
        )
      )
      .onFalse(err401.to(respond401))
  );
